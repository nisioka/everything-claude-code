+"""Per-user rate limiting middleware backed by Redis sliding window."""
+
+import time
+from typing import Callable
+
+import redis
+from fastapi import Request, Response
+
+_RATE_LIMIT_PER_MINUTE = 60
+_WINDOW_SECONDS = 60
+
+
+def make_rate_limiter(client: redis.Redis) -> Callable:
+    async def middleware(request: Request, call_next):
+        user_id = request.headers.get("x-user-id", "anon")
+        bucket_key = f"rl:{user_id}:{int(time.time() / _WINDOW_SECONDS)}"
+        count = client.incr(bucket_key)
+        if count == 1:
+            client.expire(bucket_key, _WINDOW_SECONDS * 2)
+        if count > _RATE_LIMIT_PER_MINUTE:
+            return Response(status_code=429, content="rate limit exceeded")
+        return await call_next(request)
+
+    return middleware
